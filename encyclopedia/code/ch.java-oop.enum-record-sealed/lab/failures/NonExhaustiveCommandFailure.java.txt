class NonExhaustiveCommandFailure {
    sealed interface Command permits Assign, Start, Resolve, RequestApproval {
    }
    record Assign(String id) implements Command { }
    record Start(String id) implements Command { }
    record Resolve(String id) implements Command { }
    record RequestApproval(String reason) implements Command { }

    static String summarize(Command command) {
        return switch (command) {
            case Assign assign -> "ASSIGN";
            case Start start -> "START";
            case Resolve resolve -> "RESOLVE";
        };
    }
}
